# Path B: run the playbook's gate on every push and pull request.
# Expects the playbook's tools/ folder copied into this repo as tools/.
# (Written for the playbook; check the first run's log on GitHub.)
name: check
on: [push, pull_request]
jobs:
  check:
    runs-on: ubuntu-24.04
    steps:
      - uses: actions/checkout@v4
        with:
            submodules: recursive
      - name: Toolchains
        run: |
          sudo apt-get update
          sudo apt-get install -y g++ g++-arm-linux-gnueabihf g++-aarch64-linux-gnu qemu-user lilv-utils
          pip install numpy scipy rdflib
      - name: Gate
        run: bash tools/check.sh --src plugins/nhe-ouraverb --bundle bundle/nhe-ouraverb.lv2 --test tests/test_nhe_ouraverb.py
      - name: Gate (stereo)
        run: bash tools/check_stereo.sh
      - name: Repo build
        run: |
          make NOOPT=true
          make NOOPT=true NAME=nhe-ouraverb-stereo
